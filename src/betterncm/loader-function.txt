async function loadPlugins() {
  window.loadedPlugins = loadedPlugins;
  window.betterNCMPluginErrors = { ...(window.betterNCMStartupErrors || {}) };
  if (isSafeMode()) return;
  const manifests = new Map();
  for (const [folder, devMode] of [["plugins_runtime", false], ["plugins_dev", true]]) {
    for (const path of betterncm_native.fs.readDir(folder)) {
      try {
        const manifest = JSON.parse(betterncm_native.fs.readFileText(path + "/manifest.json"));
        manifest.slug ||= manifest.name.replace(/[^a-zA-Z0-9 ]/g, "").replace(/ /g, "-");
        manifests.set(manifest.slug, { manifest, path, devMode });
      } catch (error) { window.betterNCMPluginErrors[path] = String(error); }
    }
  }
  for (const [slug, item] of manifests) for (const target of item.manifest.loadBefore || []) {
    const next = manifests.get(target);
    if (next) next.manifest.loadAfter = [...(next.manifest.loadAfter || []), slug];
  }
  const visiting = new Set(), done = new Set();
  async function load(slug) {
    if (done.has(slug)) return !!loadedPlugins[slug];
    if (visiting.has(slug)) throw new Error("Plugin dependency cycle: " + slug);
    const item = manifests.get(slug);
    if (!item) throw new Error("Missing plugin dependency: " + slug);
    const { manifest, path, devMode } = item;
    visiting.add(slug);
    try {
      if (manifest.native_plugin) throw new Error("Windows native plugin requires a Linux port");
      if (manifest["ncm-version-req"] && !window.betterNCMSatisfies(betterncm.ncm.getNCMVersion(), manifest["ncm-version-req"])) throw new Error("Unsupported NCM version: " + manifest["ncm-version-req"]);
      const requirements = [...new Set(manifest.requirements || [])];
      for (const dependency of requirements) if (!await load(dependency)) throw new Error("Dependency failed: " + dependency);
      // Ordering hints are optional, unlike requirements. A failed optional plugin
      // must not prevent an independent UI plugin from loading.
      for (const dependency of manifest.loadAfter || []) {
        if (manifests.has(dependency) && !visiting.has(dependency)) await load(dependency);
      }
      if (devMode && !manifest.noDevReload) {
        const reload = betterncm.utils.debounce(() => betterncm.reload(), 1000);
        betterncm_native.fs.watchDirectory(path, (_directory, file) => {
          if (file.endsWith(".js") || file.endsWith(".mjs") || file.endsWith("manifest.json")) reload();
        });
      }
      const main = new NCMPlugin(manifest, path, devMode);
      loadedPlugins[slug] = main;
      const AsyncFunction = Object.getPrototypeOf(async function(){}).constructor;
      const injects = [...(manifest.injects?.Main || []), ...(manifest.injects?.[location.pathname] || [])];
      for (const inject of injects) {
        const file = path + "/" + inject.file;
        const instance = new NCMInjectPlugin(main, file);
        let scriptTimeout;
        try {
          await Promise.race([new AsyncFunction("plugin", window.betterNCMTransformPluginSource ? window.betterNCMTransformPluginSource(betterncm_native.fs.readFileText(file)) : betterncm_native.fs.readFileText(file)).call(main, instance), new Promise((_, reject) => { scriptTimeout = setTimeout(() => reject(new Error("Plugin script startup timed out")), 10000); })]);
        } finally { clearTimeout(scriptTimeout); }
        instance.dispatchEvent(new CustomEvent("load", { detail: instance }));
        let timeout;
        try {
          await Promise.race([Promise.all(instance.__pendingLoads || []), new Promise((_, reject) => { timeout = setTimeout(() => reject(new Error("Plugin startup timed out")), 10000); })]);
        } finally { clearTimeout(timeout); }
        if (instance.loadError) throw instance.loadError;
        instance.finished = true;
        main.injects.push(instance);
      }
      main.finished = true;
      return true;
    } catch (error) {
      delete loadedPlugins[slug];
      window.betterNCMPluginErrors[slug] = String(error);
      console.error("[BetterNCM] Plugin failed", slug, error);
      return false;
    } finally { visiting.delete(slug); done.add(slug); }
  }
  for (const slug of manifests.keys()) { await load(slug); onPluginLoaded(loadedPlugins); }
  for (const [slug, main] of Object.entries(loadedPlugins)) {
    for (const instance of main.injects) {
      try {
        instance.dispatchEvent(new CustomEvent("allpluginsloaded", { detail: loadedPlugins }));
        let timeout;
        try {
          await Promise.race([Promise.all(instance.__pendingAllLoads || []), new Promise((_, reject) => { timeout = setTimeout(() => reject(new Error("Plugin all-plugins startup timed out")), 10000); })]);
        } finally { clearTimeout(timeout); }
        if (instance.loadError) throw instance.loadError;
      }
      catch (error) { window.betterNCMPluginErrors[slug] = String(error); }
    }
  }
  onPluginLoaded(loadedPlugins);
  setTimeout(() => window.betterNCMReportErrors?.(), 100);

}
